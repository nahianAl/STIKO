# KYC v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the 5-step /welcome wizard with one page of 9 coded dropdowns, store every answer as an append-only history row, ask invited reviewers Role only (company questions at their first project), and re-ask everyone once.

**Architecture:** Pure rules (`lib/kyc/catalog.ts`, `profile.ts`, `form.ts`, `combo.ts`) are shared by the API (authoritative) and the UI. One store module (`lib/kyc/store.ts`) owns every SQL statement for profiles and runs each submission in one `sql.transaction`. The middleware gate switches from `profile_completed_at` to `users.onboarding_version >= 2`; the profile cookie's HMAC message is versioned so old cookies stop passing.

**Tech Stack:** Next.js 14 App Router, TypeScript, Neon serverless Postgres (`sql` tagged template, `sql.transaction([...])`), Tailwind with `stiko-*` tokens, `node --test` on `.ts` files (Node 25 strips types), PGlite for local SQL checks.

**Spec:** `docs/superpowers/specs/2026-11-19-kyc-v2-design.md`. **Option lists and wording:** `docs/superpowers/specs/2026-11-19-kyc-v2-source.md`.

export function normaliseDomain(input: string): string | null {

  s = s.replace(/^[^@]*@/, '');
  s = s.replace(/:\d+$/, '');
  s = s.replace(/\.$/, '');
  s = s.replace(/^www\./, '');

/** Absent → []; anything but an array of distinct strings → null. */
function stringList(v: unknown): string[] | null {
  if (v === undefined || v === null) return [];
  if (!Array.isArray(v) || !v.every((x) => typeof x === 'string')) return null;
  if (new Set(v).size !== v.length) return null;
  return v as string[];
}

/** Checks every company answer against the catalog, in page order. */
export function validateCompany(body: unknown, email: string): Validation<CompanyAnswers, CompanyField> {
  const b = record(body);
  const fail = (field: CompanyField) => ({ ok: false as const, field });

  let companyDomain: string | null = null;
  let websiteSource: WebsiteSource = 'none';
  if (b.noWebsite !== true) {
    companyDomain = normaliseDomain(str(b.website));
    if (!companyDomain) return fail('website');
    // Typing the same domain as the prefill also reads as 'prefilled'; the two are indistinguishable.
    websiteSource = companyDomain === emailDomain(email) ? 'prefilled' : 'typed';
  }

  const industry = str(b.industry);
  if (!industryFor(industry)) return fail('industry');
  let industryOtherText: string | null = null;
  if (industry === OTHER_INDUSTRY) {
    industryOtherText = otherText(b.industryOtherText);
    if (!industryOtherText) return fail('industryOtherText');
  }

  const markets = stringList(b.markets);
  if (
    !markets || markets.length < 1 || markets.length > MAX_MARKETS ||
    !markets.every((m) => MARKETS.some((x) => x.code === m))
  ) return fail('markets');
  let marketOtherText: string | null = null;
  if (markets.includes(OTHER_MARKET)) {
    marketOtherText = otherText(b.marketOtherText);
    if (!marketOtherText) return fail('marketOtherText');
  }

  const clientModel = str(b.clientModel);
  if (!clientModelsFor(industry).some((c) => c.code === clientModel)) return fail('clientModel');

  const companySize = str(b.companySize);
  if (!SIZE_BANDS.some((s) => s.code === companySize)) return fail('companySize');

  const reviewTeamSize = str(b.reviewTeamSize);
  if (!reviewTeamBandsFor(companySize).some((r) => r.code === reviewTeamSize)) return fail('reviewTeamSize');

  const reviewerIds = stringList(b.reviewers);
  const allowed = reviewersFor(industry);
  if (!reviewerIds || !reviewerIds.every((id) => allowed.some((r) => r.id === id))) return fail('reviewers');
  const reviewerCodes = [...new Set(reviewerIds.map((id) => (REVIEWER_BY_ID.get(id) as Reviewer).code))];

  const fileTypes = stringList(b.fileTypes);
  if (!fileTypes || !fileTypes.every((c) => FILE_TYPES.some((f) => f.code === c))) return fail('fileTypes');
  let fileTypeOtherText: string | null = null;
  if (fileTypes.includes(OTHER_FILE_TYPE)) {
    fileTypeOtherText = otherText(b.fileTypeOtherText);
    if (!fileTypeOtherText) return fail('fileTypeOtherText');
  }

  return {
    ok: true,
    value: {
      companyDomain, websiteSource, industry: industry as IndustryCode, industryOtherText,
      markets, marketOtherText, clientModel, companySize, reviewTeamSize, reviewerCodes,
      fileTypes, fileTypeOtherText,
    },
  };
}

/** Role from the industry's list; null industry (Role-only page) or Other = the combined list. */
export function validateRole(body: unknown, industry: string | null): Validation<RoleAnswer, RoleField> {
  const b = record(body);
  const role = str(b.role);
  if (!rolesFor(industry).some((r) => r.code === role)) return { ok: false, field: 'role' };
  let roleOtherText: string | null = null;
  if (role === OTHER_ROLE) {
    roleOtherText = otherText(b.roleOtherText);
    if (!roleOtherText) return { ok: false, field: 'roleOtherText' };
  }
  return { ok: true, value: { roleCode: role, roleOtherText } };
}

/** Derived on the server and stored on the row; never trusted from the client. */
export function deriveCompany(a: CompanyAnswers): DerivedCompany {
  const size = SIZE_BANDS.find((s) => s.code === a.companySize)!;
  const rt = REVIEW_TEAM_BANDS.find((r) => r.code === a.reviewTeamSize)!;
  return {
    parentSegment: industryFor(a.industry)?.parent ?? null,
    primaryMarket: a.markets[0],
    sizeMin: size.min,
    sizeMax: size.max,
    rtMin: rt.min,
    rtMax: rt.max,
    icpSizeFit: size.min >= 5,
    track: size.track,
    fileTypeSupport: a.fileTypes.map((code) => ({
      code, supported: fileTypeSupported(FILE_TYPES.find((f) => f.code === code)!),
    })),
  };
}

export function deriveRole(code: string): DerivedRole {
  const role = ROLE_BY_CODE.get(code);
  const seniority = role?.seniority ?? null;
  return {
    roleFamily: role?.family ?? 'other',
    seniority,
    decisionMaker: seniority === 'owner_exec' || seniority === 'director',
    nonBuyer: seniority === 'student',
  };
}

/**
 * Which /welcome variant (spec §5). needsCompany = signed up directly or owns a
 * project. A reviewer reaches 'company' only when sent there by project creation.
 */
export function decideOnboarding(s: { roleDone: boolean; companyDone: boolean; needsCompany: boolean }): OnboardingKind {
  if (!s.roleDone) return s.needsCompany && !s.companyDone ? 'full' : 'role';
  if (!s.companyDone) return 'company';
  return 'done';
}

/** The kind the server expects, or a company edit (Settings) once onboarding is done. */
export function allowedSubmission(posted: SubmitKind, expected: OnboardingKind): boolean {
  return posted === expected || (posted === 'company' && expected === 'done');
}

/** Pure rules for components/welcome/ComboSelect.tsx. */
import { matchesSearch } from './profile.ts';

export function toggleValue(values: string[], id: string, max: number): string[] {
  if (values.includes(id)) return values.filter((v) => v !== id);
  if (values.length >= max) return values;
  return [...values, id];
}

/** Same array back when nothing moves, so React skips the update. */
export function moveValue(values: string[], from: number, to: number): string[] {
  if (from === to || from < 0 || to < 0 || from >= values.length || to >= values.length) return values;
  const next = [...values];
  const [item] = next.splice(from, 1);
  next.splice(to, 0, item);
  return next;
}

export function visibleOptions<T extends { label: string; synonyms?: readonly string[] }>(
  options: readonly T[],
  query: string
): T[] {
  return options.filter((o) => matchesSearch(o, query));
}

export function nextActive(current: number, count: number, key: 'ArrowDown' | 'ArrowUp' | 'Home' | 'End'): number {
  if (count <= 0) return 0;
  if (key === 'Home') return 0;
  if (key === 'End') return count - 1;
  if (key === 'ArrowDown') return (current + 1) % count;
  return (current - 1 + count) % count;
}

export function withIndustry(v: FormValues, code: string): FormValues {
  if (code === v.industry) return v;
  const keepClient = clientModelsFor(code).some((c) => c.code === v.clientModel);
  return {
    ...v,
    industry: code,
    industryOtherText: code === OTHER_INDUSTRY ? v.industryOtherText : '',
    role: '',
    roleOtherText: '',
    reviewers: [],
    clientModel: keepClient ? v.clientModel : '',
  };
}

export function withCompanySize(v: FormValues, code: string): FormValues {
  const fits = reviewTeamBandsFor(code).some((b) => b.code === v.reviewTeamSize);
  return { ...v, companySize: code, reviewTeamSize: fits ? v.reviewTeamSize : '' };
}

export function toRequestBody(v: FormValues, kind: SubmitKind): Record<string, unknown> {
  const company =
    kind === 'role'
      ? {}
      : {
          website: v.noWebsite ? '' : v.website,
          noWebsite: v.noWebsite,
          industry: v.industry,
          industryOtherText: v.industry === OTHER_INDUSTRY ? v.industryOtherText : undefined,
          markets: v.markets,
          marketOtherText: v.markets.includes(OTHER_MARKET) ? v.marketOtherText : undefined,
          clientModel: v.clientModel,
          companySize: v.companySize,
          reviewTeamSize: v.reviewTeamSize,
          reviewers: v.reviewers,
          fileTypes: v.fileTypes,
          fileTypeOtherText: v.fileTypes.includes(OTHER_FILE_TYPE) ? v.fileTypeOtherText : undefined,
        };
  const role =
    kind === 'company'
      ? {}
      : { role: v.role, roleOtherText: v.role === OTHER_ROLE ? v.roleOtherText : undefined, marketingConsent: v.marketingConsent };
  return { kind, ...company, ...role };
}

export function firstInvalidField(v: FormValues, kind: SubmitKind): CompanyField | RoleField | null {
  const body = toRequestBody(v, kind);
  if (kind !== 'role') {
    const c = validateCompany(body, '');
    if (!c.ok) return c.field;
  }
  if (kind !== 'company') {
    const r = validateRole(body, kind === 'role' ? null : v.industry);
    if (!r.ok) return r.field;
  }
  return null;
}

    | {
        email: string; workos_user_id: string | null; onboarding_version: number | null;
        signup_type: string | null; profile_completed_at: string | null;
        company_version: number | null; owns_project: boolean;
      }
    | undefined;
  if (!r) return null;
  const roleDone = Number(r.onboarding_version ?? 0) >= ONBOARDING_VERSION;
  const companyDone = Number(r.company_version ?? 0) >= ONBOARDING_VERSION;
  // Users from before signup_type existed (NULL) are treated as direct signups.
  const needsCompany = r.signup_type !== 'invited_reviewer' || r.owns_project;
  return {
    kind: decideOnboarding({ roleDone, companyDone, needsCompany }),
    roleDone, companyDone, needsCompany,
    email: r.email, workosUserId: r.workos_user_id, profileCompletedAt: r.profile_completed_at,
  };
}

export async function companyProfileDone(userId: string): Promise<boolean> {
  const rows = await sql`
    SELECT cp.onboarding_version
    FROM workspaces w JOIN company_profiles cp ON cp.id = w.current_profile_id
    WHERE w.owner_user_id = ${userId}
  `;
  return Number(rows[0]?.onboarding_version ?? 0) >= ONBOARDING_VERSION;
}

/** The current v2 profile as form values, for Settings → Company. */
export async function currentCompanyForm(userId: string): Promise<Partial<FormValues> | null> {
  const rows = await sql`
    SELECT cp.*,
      ARRAY(SELECT market_code FROM company_profile_markets m WHERE m.profile_id = cp.id ORDER BY pick_order) AS markets,
      ARRAY(SELECT reviewer_code FROM company_profile_reviewers r WHERE r.profile_id = cp.id) AS reviewer_codes,
      ARRAY(SELECT file_type_code FROM company_profile_file_types f WHERE f.profile_id = cp.id ORDER BY file_type_code) AS file_types
    FROM workspaces w JOIN company_profiles cp ON cp.id = w.current_profile_id
    WHERE w.owner_user_id = ${userId} AND cp.onboarding_version >= ${ONBOARDING_VERSION}
  `;
  const p = rows[0] as Record<string, unknown> | undefined;
  if (!p) return null;
  const industry = p.industry as string;
  // Codes back to option ids: the first option in this industry's list with that code.
  const options = reviewersFor(industry);
  const reviewers = (p.reviewer_codes as string[])
    .map((code) => options.find((o) => o.code === code)?.id)
    .filter((id): id is string => Boolean(id));
  return {
    website: (p.company_domain as string | null) ?? '',
    noWebsite: p.website_source === 'none',
    industry,
    industryOtherText: (p.industry_other_text as string | null) ?? '',
    markets: p.markets as string[],
    marketOtherText: (p.market_other_text as string | null) ?? '',
    clientModel: p.client_model as string,
    companySize: p.company_size as string,
    reviewTeamSize: p.review_team_size as string,
    reviewers,
    fileTypes: p.file_types as string[],
    fileTypeOtherText: (p.file_type_other_text as string | null) ?? '',
  };
}

/** One transaction per submission (spec §7). */
export async function saveOnboarding(input: SaveInput): Promise<{ profileId: string | null }> {
  const { userId, workspaceId, company, role, source, consent, country, region, businessEmail } = input;
  const V = ONBOARDING_VERSION;
  const profileId = company ? uuidv4() : null;
  const queries = [];

  if (company && profileId) {
    const d = deriveCompany(company);
    queries.push(
      sql`
        INSERT INTO company_profiles (
          id, workspace_id, submitted_by_user_id, onboarding_version, source,
          company_domain, website_source, industry, industry_other_text, parent_segment, client_model,
          company_size, size_min, size_max, review_team_size, rt_min, rt_max,
          primary_market, market_other_text, file_type_other_text, icp_size_fit
        ) VALUES (
          ${profileId}, ${workspaceId}, ${userId}, ${V}, ${source},
          ${company.companyDomain}, ${company.websiteSource}, ${company.industry}, ${company.industryOtherText},
          ${d.parentSegment}, ${company.clientModel},
          ${company.companySize}, ${d.sizeMin}, ${d.sizeMax}, ${company.reviewTeamSize}, ${d.rtMin}, ${d.rtMax},
          ${d.primaryMarket}, ${company.marketOtherText}, ${company.fileTypeOtherText}, ${d.icpSizeFit}
        )
      `,
      sql`
        INSERT INTO company_profile_markets (profile_id, market_code, pick_order)
        SELECT ${profileId}, m, o FROM unnest(${company.markets}::text[]) WITH ORDINALITY AS t(m, o)
      `,
      sql`
        INSERT INTO company_profile_reviewers (profile_id, reviewer_code)
        SELECT ${profileId}, c FROM unnest(${company.reviewerCodes}::text[]) AS t(c)
      `,
      sql`
        INSERT INTO company_profile_file_types (profile_id, file_type_code, supported_at_signup)
        SELECT ${profileId}, c, s
        FROM unnest(${d.fileTypeSupport.map((f) => f.code)}::text[], ${d.fileTypeSupport.map((f) => f.supported)}::boolean[]) AS t(c, s)
      `,
      sql`
        UPDATE workspaces SET
          current_profile_id = ${profileId}, company_domain = ${company.companyDomain},
          country = COALESCE(${country}, country), region = COALESCE(${region}, region),
          is_business_email = ${businessEmail}
        WHERE id = ${workspaceId}
      `
    );
  }

  if (role) {
    const r = deriveRole(role.roleCode);
    queries.push(
      sql`
        INSERT INTO user_roles (
          id, user_id, onboarding_version, source, role_code, role_other_text, role_family, seniority,
          decision_maker, non_buyer
        ) VALUES (
          ${uuidv4()}, ${userId}, ${V}, ${source}, ${role.roleCode}, ${role.roleOtherText}, ${r.roleFamily},
          ${r.seniority}, ${r.decisionMaker}, ${r.nonBuyer}
        )
      `,
      sql`
        UPDATE users SET
          role = ${role.roleCode},
          onboarding_version = GREATEST(COALESCE(onboarding_version, 0), ${V}),
          profile_completed_at = COALESCE(profile_completed_at, now()),
          is_business_email = ${businessEmail}
        WHERE id = ${userId}
      `
    );
  }

  if (consent) {
    queries.push(sql`
      UPDATE users SET
        marketing_consent = ${consent.marketing},
        marketing_consent_at = CASE WHEN ${consent.marketing} THEN now() ELSE NULL END,
        marketing_consent_method = CASE WHEN ${consent.marketing} THEN 'welcome_checkbox' ELSE NULL END,
        analytics_consent = COALESCE(${consent.analytics}, analytics_consent, ${consent.pendingIfRequired})
      WHERE id = ${userId}
    `);
  }

  await sql.transaction(queries);
  return { profileId };
}

export async function GET() {

export async function POST(request: NextRequest) {
  const session = await auth();
  if (!session?.user?.id) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  const userId = session.user.id;

  const body = await request.json().catch(() => null);
  const kind = (body?.kind ?? '') as SubmitKind;
  if (!KINDS.includes(kind)) return NextResponse.json({ error: 'invalid', field: 'kind' }, { status: 400 });

  const state = await onboardingState(userId);
  if (!state) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  if (!allowedSubmission(kind, state.kind)) {
    return NextResponse.json({ error: 'wrong_kind', kind: state.kind }, { status: 409 });
  }

  let company: CompanyAnswers | null = null;
  if (kind !== 'role') {
    const c = validateCompany(body, state.email);
    if (!c.ok) return NextResponse.json({ error: 'invalid', field: c.field }, { status: 400 });
    company = c.value;
  }
  let role: RoleAnswer | null = null;
  if (kind !== 'company') {
    const r = validateRole(body, kind === 'role' ? null : company!.industry);
    if (!r.ok) return NextResponse.json({ error: 'invalid', field: r.field }, { status: 400 });
    role = r.value;
  }

  const country = request.headers.get('x-vercel-ip-country');
  const region = request.headers.get('x-vercel-ip-country-region');
  const marketing = body?.marketingConsent === true;
  const source =
    kind !== 'company' ? 'onboarding' : body?.source === 'settings' ? 'settings' : 'project_gate';
  const businessEmail = isBusinessEmail(state.email);

  const workspaceId = await ensureWorkspace(userId);
  await saveOnboarding({
    userId, workspaceId, company, role, source,
    // A user imported before 019 has analytics_consent NULL ("never asked, allowed"). In a
    // consent country that is wrong: until they answer the banner they are 'pending'.
    consent: kind === 'company' ? null : {
      marketing,
      analytics: consentValue(cookies().get(CONSENT_COOKIE)?.value),
      pendingIfRequired: consentRequired(country) ? 'pending' : null,
    },
    country, region, businessEmail,
  });

  // Codes only: "Other" free text never leaves Postgres (plan Global Constraints).
  const resubmitted = state.profileCompletedAt !== null;
  if (company) {
    const d = deriveCompany(company);
    const r = role ? deriveRole(role.roleCode) : null;
    await track('company_profile_completed', {
      userId,
      properties: {
        kind, source, industry: company.industry, parent_segment: d.parentSegment,

        website_source: company.websiteSource, track: d.track, country, region, resubmitted,
        ...(role && r ? { role: role.roleCode, seniority: r.seniority, role_family: r.roleFamily } : {}),
        ...(kind === 'full' ? { marketing_consent: marketing } : {}),
      },
      set: {
        is_business_email: businessEmail,
        ...(role && r ? { role: role.roleCode, seniority: r.seniority, role_family: r.roleFamily, decision_maker: r.decisionMaker, non_buyer: r.nonBuyer } : {}),
        ...(kind === 'full' ? { marketing_consent: marketing } : {}),
      },
    });
  } else if (role) {
    const r = deriveRole(role.roleCode);
    await track('reviewer_role_completed', {
      userId,
      properties: { role: role.roleCode, seniority: r.seniority, role_family: r.roleFamily, marketing_consent: marketing, resubmitted },
      set: { role: role.roleCode, seniority: r.seniority, role_family: r.roleFamily, decision_maker: r.decisionMaker, non_buyer: r.nonBuyer, marketing_consent: marketing },
    });
  }
  await identifyWorkspace(userId);

  const redirectPath = safeCallbackUrl(typeof body?.callbackUrl === 'string' ? body.callbackUrl : null);
  const res = NextResponse.json({ redirectPath });

  // The gate's cookie subject is the WorkOS user id under AUTH_PROVIDER=workos and the

  const secret = process.env.PROFILE_COOKIE_SECRET;

  if (role && secret && subject) {
    res.headers.append(
      'Set-Cookie',
      profileCookieHeader(await signProfileCookie(subject, secret), process.env.NODE_ENV === 'production')
    );
  }
  return res;
}

interface SingleProps extends CommonProps { multiple?: false; value: string; onChange: (v: string) => void }
interface MultiProps extends CommonProps {
  multiple: true;
  value: string[];
  onChange: (v: string[]) => void;
  max?: number;
  /** Label the first tag "Primary" and let tags be reordered (Market). */
  primaryTag?: boolean;
}

/**
 * A dropdown with optional search, group headers and removable tags (spec §6).
 * Keyboard: ArrowUp/Down move, Enter picks, Escape closes this dropdown only,
 * Tab closes and moves on. Tags reorder by drag or by ArrowLeft/Right when focused.
 */
export default function ComboSelect(props: SingleProps | MultiProps) {
  const { id, labelId, options, placeholder = 'Select…', searchable = false, invalid = false, disabled = false } = props;
  const listId = useId();
  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState('');
  const [active, setActive] = useState(0);
  const [dragFrom, setDragFrom] = useState<number | null>(null);
  const rootRef = useRef<HTMLDivElement>(null);
  const triggerRef = useRef<HTMLButtonElement>(null);
  const searchRef = useRef<HTMLInputElement>(null);

  const selected = props.multiple ? props.value : props.value ? [props.value] : [];
  const max = props.multiple ? props.max ?? Infinity : 1;
  const shown = useMemo(() => visibleOptions(options, query), [options, query]);
  const byId = useMemo(() => new Map(options.map((o) => [o.id, o])), [options]);

  useEffect(() => {
    if (!open) setQuery('');
    else if (searchable) searchRef.current?.focus();
  }, [open, searchable]);
  useEffect(() => setActive(0), [query, open]);
  useEffect(() => {
    if (open) document.getElementById(`${listId}-${active}`)?.scrollIntoView({ block: 'nearest' });
  }, [active, open, listId]);
  useEffect(() => {
    if (!open) return;
    const onDown = (e: PointerEvent) => {
      if (!rootRef.current?.contains(e.target as Node)) setOpen(false);
    };
    document.addEventListener('pointerdown', onDown);
    return () => document.removeEventListener('pointerdown', onDown);
  }, [open]);

  const close = () => {
    setOpen(false);
    triggerRef.current?.focus();
  };

  const choose = (optId: string) => {
    if (props.multiple) props.onChange(toggleValue(props.value, optId, max));
    else {
      props.onChange(optId);
      close();
    }
  };

  const onKeyDown = (e: React.KeyboardEvent) => {
    const nav = e.key === 'ArrowDown' || e.key === 'ArrowUp' || (!searchable && (e.key === 'Home' || e.key === 'End'));
    if (nav) {
      e.preventDefault();
      if (!open) setOpen(true);
      else setActive((a) => nextActive(a, shown.length, e.key as 'ArrowDown' | 'ArrowUp' | 'Home' | 'End'));
      return;
    }
    if (e.key === 'Enter' && open) {
      e.preventDefault();
      const o = shown[active];
      if (o) choose(o.id);
      return;
    }
    if (e.key === 'Escape' && open) {
      // Close this dropdown only, never the page or a dialog around it.
      e.preventDefault();
      e.stopPropagation();
      close();
      return;
    }
    if (e.key === 'Tab') setOpen(false);
  };

  const summary = props.multiple
    ? props.value.length >= max ? `${max} selected` : placeholder
    : byId.get(props.value)?.label ?? placeholder;
  const hasValue = props.multiple ? false : Boolean(props.value);

  let lastGroup: string | undefined;
  const activeId = open && shown[active] ? `${listId}-${active}` : undefined;

  return (
    <div ref={rootRef} className="relative">
      <div
        className={`flex min-h-[40px] flex-wrap items-center gap-[6px] rounded-[10px] border bg-white px-[10px] py-[5px] ${
          invalid ? 'border-stiko-danger' : 'border-stiko-border'
        } ${disabled ? 'opacity-50' : ''} focus-within:shadow-stiko-focus`}
      >
        {props.multiple &&
          props.value.map((v, i) => {
            const multi = props as MultiProps;
            const reorder = Boolean(multi.primaryTag) && multi.value.length > 1;
            return (
              <span
                key={v}
                tabIndex={reorder ? 0 : -1}
                draggable={reorder}
                onDragStart={() => setDragFrom(i)}
                onDragOver={(e) => reorder && e.preventDefault()}
                onDrop={() => {
                  if (dragFrom !== null) multi.onChange(moveValue(multi.value, dragFrom, i));
                  setDragFrom(null);
                }}
                onKeyDown={(e) => {
                  if (!reorder) return;
                  if (e.key === 'ArrowLeft') { e.preventDefault(); multi.onChange(moveValue(multi.value, i, i - 1)); }
                  if (e.key === 'ArrowRight') { e.preventDefault(); multi.onChange(moveValue(multi.value, i, i + 1)); }
                }}
                aria-label={reorder ? `${byId.get(v)?.label ?? v}${i === 0 ? ', primary' : ''}. Use left and right arrows to reorder.` : undefined}
                className="inline-flex items-center gap-[5px] rounded-full bg-stiko-app px-[9px] py-[3px] text-[12.5px] font-semibold text-stiko-ink outline-none focus-visible:shadow-stiko-focus"
              >
                {multi.primaryTag && i === 0 && (
                  <span className="rounded-full bg-gradient-to-br from-[#8094F5] to-[#5B60FF] px-[6px] text-[10.5px] font-bold text-white">
                    Primary
                  </span>
                )}
                {byId.get(v)?.label ?? v}
                <button
                  type="button"
                  disabled={disabled}
                  onClick={() => multi.onChange(multi.value.filter((x) => x !== v))}
                  aria-label={`Remove ${byId.get(v)?.label ?? v}`}
                  className="text-stiko-muted hover:text-stiko-ink"
                >
                  ×
                </button>
              </span>
            );
          })}
        <button
          ref={triggerRef}
          id={id}
          type="button"
          role="combobox"
          aria-labelledby={labelId}
          aria-expanded={open}
          aria-controls={listId}
          aria-haspopup="listbox"
          aria-invalid={invalid || undefined}
          aria-activedescendant={searchable ? undefined : activeId}
          disabled={disabled}
          onClick={() => setOpen((o) => !o)}
          onKeyDown={onKeyDown}
          className={`min-w-[120px] flex-1 py-[4px] text-left text-[13.5px] outline-none ${
            hasValue ? 'text-stiko-ink' : 'text-stiko-faint'
          }`}
        >
          {summary}
          <span aria-hidden="true" className="float-right text-stiko-muted">▾</span>
        </button>
      </div>

      {open && (
        <div className="absolute z-20 mt-1 w-full overflow-hidden rounded-[10px] border border-stiko-border bg-white shadow-lg">
          {searchable && (
            <input
              ref={searchRef}
              type="text"
              role="searchbox"
              aria-label="Search"
              aria-controls={listId}
              aria-activedescendant={activeId}
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              onKeyDown={onKeyDown}
              placeholder="Type to search"
              className="w-full border-b border-stiko-border px-3 py-2 text-[13.5px] outline-none"
            />
          )}
          <ul
            id={listId}
            role="listbox"
            aria-labelledby={labelId}
            aria-multiselectable={props.multiple || undefined}
            className="max-h-[280px] overflow-y-auto py-1"
          >
            {shown.length === 0 && <li className="px-3 py-2 text-[13px] text-stiko-muted">No matches</li>}
            {shown.map((o, i) => {
              const header = o.group && o.group !== lastGroup ? o.group : null;
              lastGroup = o.group;
              const isSelected = selected.includes(o.id);
              const isDisabled = !isSelected && props.multiple === true && selected.length >= max;
              return (
                <React.Fragment key={o.id}>
                  {header && (
                    <li role="presentation" className="px-3 pb-1 pt-2 text-[11px] font-bold uppercase tracking-wide text-stiko-faint">
                      {header}
                    </li>
                  )}
                  <li
                    id={`${listId}-${i}`}
                    role="option"
                    aria-selected={isSelected}
                    aria-disabled={isDisabled || undefined}
                    onMouseDown={(e) => e.preventDefault()}
                    onMouseEnter={() => setActive(i)}
                    onClick={() => !isDisabled && choose(o.id)}
                    className={`flex cursor-pointer items-start gap-2 px-3 py-[7px] text-[13.5px] ${
                      i === active ? 'bg-stiko-app' : ''
                    } ${isDisabled ? 'cursor-not-allowed opacity-40' : ''}`}
                  >
                    {props.multiple && (
                      <span aria-hidden="true" className={`mt-[3px] inline-block h-[14px] w-[14px] shrink-0 rounded-[4px] border ${
                        isSelected ? 'border-stiko-primary bg-stiko-primary' : 'border-stiko-border'
                      }`} />
                    )}
                    <span className="min-w-0">
                      <span className={`block ${isSelected ? 'font-bold text-stiko-ink' : 'text-stiko-ink'}`}>{o.label}</span>
                      {o.helper && <span className="block text-[12px] text-stiko-muted">{o.helper}</span>}
                    </span>
                  </li>
                </React.Fragment>
              );
            })}
          </ul>
        </div>
      )}
    </div>
  );
}
```

    fetch('/api/welcome')
      .then(async (r) => {
        if (r.status === 401) return loginAgain();
        if (!r.ok) throw new Error(String(r.status));
        const data = (await r.json()) as WelcomeState;

              {COPY[kind].title}

/** The 409 from POST /api/projects when the company questions are still unanswered (spec §5). */
export const COMPANY_PROFILE_REQUIRED = 'company_profile_required';

export async function isCompanyProfileRequired(res: Response): Promise<boolean> {
  if (res.status !== 409) return false;
  const data = await res.clone().json().catch(() => null);
  return data?.code === COMPANY_PROFILE_REQUIRED;
}

export function companyProfilePath(returnTo: string): string {
  return `/welcome?${new URLSearchParams({ callbackUrl: returnTo }).toString()}`;
}

'use client';

import { useEffect, useState } from 'react';
import { SettingsCard, SettingsShell } from '@/components/settings/SettingsShell';
import { ACCOUNT_RAIL } from '@/components/settings/accountRail';
import { useToast } from '@/components/ui/Toast';
import OnboardingForm from '@/components/welcome/OnboardingForm';
import { EMPTY_FORM, FIELD_ERRORS, toRequestBody, type FormValues } from '@/lib/kyc/form';

/** The company answers, editable. Each save adds a history row (spec §7). */
export default function CompanySettings() {
  const { toast } = useToast();
  const [initial, setInitial] = useState<FormValues | null>(null);
  const [prefill, setPrefill] = useState('');
  const [formKey, setFormKey] = useState(0);
  const [saving, setSaving] = useState(false);
  const [serverError, setServerError] = useState<{ field: string | null; message: string } | null>(null);

  useEffect(() => {
    fetch('/api/welcome')
      .then((r) => (r.ok ? r.json() : null))
      .then((s) => {
        setPrefill(s?.prefillWebsite ?? '');
        setInitial({ ...EMPTY_FORM, ...(s?.current ?? {}) });
      })
      .catch(() => setInitial(EMPTY_FORM));
  }, []);

  const save = async (values: FormValues) => {
    setSaving(true);
    setServerError(null);
    try {
      const res = await fetch('/api/welcome', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ ...toRequestBody(values, 'company'), source: 'settings' }),
      });
      const data = await res.json().catch(() => ({}));
      setSaving(false);
      if (res.ok) {
        toast('Company profile saved');
        setInitial(values);
        setFormKey((k) => k + 1);
        return;
      }
      const field = typeof data.field === 'string' ? data.field : null;
      setServerError({
        field,
        message: (field && FIELD_ERRORS[field as keyof typeof FIELD_ERRORS]) || 'Could not save. Try again.',
      });
    } catch {
      setSaving(false);
      setServerError({ field: null, message: 'Could not save. Check your connection and try again.' });
    }
  };

  return (
    <SettingsShell crumbs={[{ label: 'Settings' }, { label: 'Company' }]} railLabel="Account" items={ACCOUNT_RAIL} active="company">
      <SettingsCard heading="Company" description="What your company does and who reviews your work. Changes keep a history.">
        {initial && (
          <OnboardingForm
            key={formKey}
            variant="company"
            initial={initial}
            prefillWebsite={prefill}
            submitLabel="Save changes"
            showMarketing={false}
            saving={saving}
            serverError={serverError}
            onSubmit={save}
          />
        )}
      </SettingsCard>
    </SettingsShell>
  );
}
```
